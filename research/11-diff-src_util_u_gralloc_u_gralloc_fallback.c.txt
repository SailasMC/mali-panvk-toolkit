+/* task-11 ①b（对应 09 §7 方案③b / MR !43659 "infer LINEAR modifier for tightly
+ * packed fallback buffers"）：当厂商 mapper 元数据不可用、只能拿到
+ * modifier = DRM_FORMAT_MOD_INVALID 时，对"紧凑打包的线性"缓冲推断 LINEAR。
+ *
+ * 为什么安全（不是猜）：
+ *  - 能走到这里，说明 fourcc 已由 get_fourcc_from_hal_format(hal_format) 解析成功
+ *    （YUV/未知格式在上面已经返回），行距 strides[0] 已知且 > 0；
+ *  - 若该 dma_buf 的总大小恰为行距的整数倍（st_size % pitch == 0），则其布局
+ *    只可能是逐行紧凑排布（行距已含 gralloc 自己的对齐填充），即 LINEAR；
+ *  - 其余情况（无法 fstat、大小对不齐、多平面、非零 offset、已有真实 modifier）
+ *    一律拒绝并保持原来的失败语义。
+ * panvk 对 DRM_FORMAT_MOD_INVALID 没有 handler（pan_mod_get_handler(INVALID)
+ * == NULL），所以不给 LINEAR 就必然在 pan_image_layout_init 里崩。
+ *
+ * 开关：PANVK_GRALLOC_NO_INFER_LINEAR=1 一键退回旧行为（拒绝并上抛）。
+ */
+static bool
+panvk_infer_linear_modifier(struct u_gralloc_buffer_handle *hnd,
+                            struct u_gralloc_buffer_basic_info *out,
+                            const char *why)
+{
+   if (debug_get_bool_option("PANVK_GRALLOC_NO_INFER_LINEAR", false))
+      return false;
+
+   if (!out || out->modifier != DRM_FORMAT_MOD_INVALID || out->num_planes != 1 ||
+       out->offsets[0] != 0 || out->strides[0] <= 0)
+      return false;
+   if (!hnd || !hnd->handle || hnd->handle->numFds < 1 ||
+       hnd->handle->data[0] < 0)
+      return false;
+
+   struct stat st;
+   if (fstat(hnd->handle->data[0], &st) != 0 || st.st_size <= 0)
+      return false;
+
+   uint64_t size = (uint64_t)st.st_size;
+   if (size % (uint64_t)out->strides[0] != 0)
+      return false;
+
+   uint64_t implied_h = size / (uint64_t)out->strides[0];
+   if (implied_h == 0 || implied_h > 65536)
+      return false;
+
+   out->modifier = DRM_FORMAT_MOD_LINEAR;
+   if (out->alloc_size == 0)
+      out->alloc_size = size;
+   if (out->layer_count == 0)
+      out->layer_count = 1;
+
+   mesa_logi("[PANVK-LINEAR-INFER] %s: fourcc=0x%08x pitch=%d size=%llu "
+             "implied_h=%llu -> DRM_FORMAT_MOD_LINEAR",
+             why, out->drm_fourcc, out->strides[0],
+             (unsigned long long)size, (unsigned long long)implied_h);
+   return true;
+}
